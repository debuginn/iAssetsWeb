{{- $trust := .trust | default dict -}} {{- $institutions := .institutions | default dict -}} {{- $lang := lower (replace (.lang | default "en") "_" "-") -}} {{- $aliases := dict "zh-cn" "zh" "zh-hans" "zh" "zh-hans-cn" "zh" "zh-tw" "zh-hant" "zh-hk" "zh-hant" "zh-mo" "zh-hant" "zh-hant-tw" "zh-hant" "zh-hant-hk" "zh-hant" "zh-hant-mo" "zh-hant" "tw" "zh-hant" "hk" "zh-hant" "mo" "zh-hant" "sg" "zh-hant" "en-us" "en" "en-gb" "en" "ja-jp" "ja" "ko-kr" "ko" -}} {{- $lang = index $aliases $lang | default $lang -}} {{- with $trust.tracks -}} {{- $tracks := dict -}} {{- range $track, $items := . -}} {{- $localizedItems := slice -}} {{- range $item := $items -}} {{- /* Match the institution, not its position or already-translated label. */ -}} {{- $src := $item.src | default "" -}} {{- $names := index $institutions $src | default dict -}} {{- $english := strings.TrimSpace ($names.en | default "") -}} {{- if not $english -}} {{- errorf "Trust institution %q requires an English name in data/trust_institutions.json" $src -}} {{- end -}} {{- $name := strings.TrimSpace (index $names $lang | default "") -}} {{- if not $name -}} {{- $name = $english -}} {{- end -}} {{- /* Keep a published alt, including its 标志/logo/ロゴ/로고 suffix. A few English alts are not the short label plus " logo" (HSBC China logo, Standard Chartered China logo, Interactive Brokers logo, FlyBay logo). Only synthesize an alt when the page did not provide one. */ -}} {{- $alt := strings.TrimSpace ($item.alt | default "") -}} {{- if not $alt -}} {{- $suffixes := dict "zh" "标志" "zh-hant" "標誌" "ja" "のロゴ" "ko" " 로고" "en" " logo" -}} {{- $alt = printf "%s%s" $name (index $suffixes $lang | default " logo") -}} {{- end -}} {{- $localizedItems = $localizedItems | append (merge $item (dict "name" $name "alt" $alt)) -}} {{- end -}} {{- $tracks = merge $tracks (dict $track $localizedItems) -}} {{- end -}} {{- $trust = merge $trust (dict "tracks" $tracks) -}} {{- end -}} {{- return $trust -}}
